#### **Module 2: Shade Generation (CRITICAL)**
```javascript
/**
 * Generate 11 Tailwind-style shades from a single hex color
 * @param {string} baseHex - Input color (maps to shade 500)
 * @returns {Object} - { 50: '#...', 100: '#...', ..., 950: '#...' }
 */
function generateShades(baseHex) {
  const oklch = hexToOklch(baseHex);
  
  // Lightness targets for each shade (perceptually even)
  const lightnessMap = {
    50: 0.97,   // Near white
    100: 0.93,
    200: 0.85,
    300: 0.75,
    400: 0.63,
    500: oklch.L,  // Base color
    600: oklch.L * 0.82,
    700: oklch.L * 0.68,
    800: oklch.L * 0.54,
    900: oklch.L * 0.42,
    950: oklch.L * 0.28,  // Near black
  };
  
  // Chroma adjustments (saturation boost for very light/dark shades)
  const chromaMap = {
    50: oklch.C * 0.3,   // Desaturated
    100: oklch.C * 0.5,
    200: oklch.C * 0.7,
    300: oklch.C * 0.85,
    400: oklch.C * 0.95,
    500: oklch.C,         // Original saturation
    600: oklch.C * 1.05,
    700: oklch.C * 1.08,
    800: oklch.C * 1.05,
    900: oklch.C * 0.95,
    950: oklch.C * 0.7,   // Desaturated
  };
  
  const shades = {};
  for (const [shade, targetL] of Object.entries(lightnessMap)) {
    const targetC = chromaMap[shade];
    shades[shade] = oklchToHex(targetL, targetC, oklch.H);
  }
  
  return shades;
}
```

#### **Module 3: State Management**
```javascript
// State (mutable)
let colors = [];  // Array of { id, name, baseHex, shades: {...} }
let history = []; // Undo/redo stack
let historyIndex = -1;
let currentPreviewTab = 'cards';
let currentExportTab = 'tailwind3';

// State mutations
function addColor(name = 'primary', baseHex = '#3b82f6') {
  const id = Date.now() + Math.random();
  const shades = generateShades(baseHex);
  colors.push({ id, name, baseHex, shades });
  saveState();
  render();
}

function updateColorBase(id, newBaseHex) {
  const color = colors.find(c => c.id === id);
  if (color) {
    color.baseHex = newBaseHex;
    color.shades = generateShades(newBaseHex);
    saveState();
    render();
  }
}

function updateShade(id, shadeKey, newHex) {
  const color = colors.find(c => c.id === id);
  if (color) {
    color.shades[shadeKey] = newHex;
    // Mark as manually edited (don't regenerate this shade)
    color.manualShades = color.manualShades || {};
    color.manualShades[shadeKey] = true;
    saveState();
    render();
  }
}

function removeColor(id) {
  colors = colors.filter(c => c.id !== id);
  saveState();
  render();
}

function saveState() {
  const state = JSON.stringify(colors);
  if (historyIndex < history.length - 1) {
    history = history.slice(0, historyIndex + 1);
  }
  history.push(state);
  historyIndex++;
  if (history.length > 50) {
    history.shift();
    historyIndex--;
  }
}

function undo() {
  if (historyIndex > 0) {
    historyIndex--;
    colors = JSON.parse(history[historyIndex]);
    render();
  }
}

function redo() {
  if (historyIndex < history.length - 1) {
    historyIndex++;
    colors = JSON.parse(history[historyIndex]);
    render();
  }
}
```

function generateTailwind3(colors) {
  let output = '// Tailwind CSS v3 Config\n';
  output += 'module.exports = {\n';
  output += '  theme: {\n';
  output += '    extend: {\n';
  output += '      colors: {\n';
  
  colors.forEach(color => {
    output += `        '${color.name}': {\n`;
    for (const [shade, hex] of Object.entries(color.shades)) {
      output += `          ${shade}: '${hex}',\n`;
    }
    output += '        },\n';
  });
  
  output += '      },\n';
  output += '    },\n';
  output += '  },\n';
  output += '};\n';
  
  return output;
}

function generateTailwind4(colors) {
  let output = '/* Tailwind CSS v4 with OKLCH */\n';
  output += '@theme {\n';
  
  colors.forEach(color => {
    for (const [shade, hex] of Object.entries(color.shades)) {
      const oklch = hexToOklch(hex);
      const L = (oklch.L * 100).toFixed(2);
      const C = oklch.C.toFixed(4);
      const H = oklch.H.toFixed(2);
      
      output += `  --color-${color.name}-${shade}: oklch(${L}% ${C} ${H});\n`;
    }
  });
  
  output += '}\n';
  return output;
}

function generateCSS(colors) {
  let output = '/* CSS Variables */\n';
  output += ':root {\n';
  
  colors.forEach(color => {

      output += `  --${color.name}-${shade}: ${hex};\n`;

  output += '}\n';
  return output;
}

function generateFigma(colors, mode = 'light') {
  // W3C Design Tokens format (JSON)
  const tokens = {
    "$schema": "https://design-tokens.github.io/community-group/format/",

## 3. Shade Generation Algorithm (CRITICAL)

### 3.1 The Challenge

**Goal:** Generate 11 perceptually even shades (50-950) from a single base color (maps to shade 500).

**Requirements:**
- Shades must feel evenly distributed to the human eye
- Lighter shades (50-400) should get progressively lighter
- Darker shades (600-950) should get progressively darker
- Hue should remain constant (or nearly so)
- Saturation should adjust to maintain vibrancy

**Why OKLCH?**
- Perceptual color space (matches human vision)
- Lightness (L) is perceptually uniform (unlike HSL)
- Chroma (C) represents saturation in a consistent way
- Hue (H) is stable across lightness changes

### 3.2 Algorithm Approach

#### **Step 1: Convert Base Color to OKLCH**
```javascript
const baseHex = '#3b82f6'; // Example: Tailwind Blue 500
const oklch = hexToOklch(baseHex);
// Result: { L: 0.56, C: 0.21, H: 248 }
```

#### **Step 2: Define Lightness Targets**

**Strategy:** Distribute lightness (L) evenly across the scale

**Tailwind Reference Analysis:**
- Shade 50: Very light (L ≈ 0.97)
- Shade 100: Light (L ≈ 0.93)
- Shade 200-400: Mid-light (gradual decrease)
- Shade 500: Base color (original L)
- Shade 600-900: Mid-dark to dark
- Shade 950: Very dark (L ≈ base L × 0.25-0.3)

**Implementation:**
```javascript
function getLightnessTarget(shade, baseL) {
  const targets = {
    50: 0.97,   // Fixed: near white
    100: 0.93,  // Fixed: very light
    200: 0.85,  // Fixed
    300: 0.75,  // Fixed
    400: 0.63,  // Fixed
    500: baseL, // Original lightness
    600: baseL * 0.82,  // Relative to base
    700: baseL * 0.68,
    800: baseL * 0.54,
    900: baseL * 0.42,
    950: baseL * 0.28,  // Very dark
  };
  
  return targets[shade];
}
```

**Why This Works:**
- Shades 50-400 are absolute (consistent across all colors)
- Shades 600-950 are relative (scale with base color darkness)
- This mimics Tailwind's actual color scales

#### **Step 3: Adjust Chroma (Saturation)**

**Problem:** Fixed chroma across all lightness levels looks unnatural
- Very light colors (50-100) need less saturation (avoid neon pastels)
- Very dark colors (900-950) need less saturation (avoid muddy darks)
- Mid-range (400-700) can handle higher saturation

**Solution:** Chroma curve based on shade

```javascript
function getChromaMultiplier(shade) {
  const multipliers = {
    50: 0.3,   // Highly desaturated
    100: 0.5,
    200: 0.7,
    300: 0.85,
    400: 0.95,
    500: 1.0,  // Original chroma
    600: 1.05, // Slight boost
    700: 1.08, // Peak saturation
    800: 1.05,
    900: 0.95,
    950: 0.7,  // Desaturated dark
  };
  
  return multipliers[shade];
}
```

**Rationale:**
- Light shades (50-100): Reduce chroma → soft pastels
- Dark shades (900-950): Reduce chroma → deep, rich darks (not muddy)
- Mid-range (600-700): Boost chroma → vibrant colors

#### **Step 4: Generate All Shades**

```javascript
function generateShades(baseHex) {
  const baseOklch = hexToOklch(baseHex);
  const shades = {};
  
  const shadeKeys = [50, 100, 200, 300, 400, 500, 600, 700, 800, 900, 950];
  
  for (const shade of shadeKeys) {
    const targetL = getLightnessTarget(shade, baseOklch.L);
    const chromaMult = getChromaMultiplier(shade);
    const targetC = baseOklch.C * chromaMult;
    const targetH = baseOklch.H; // Keep hue constant
    
    shades[shade] = oklchToHex(targetL, targetC, targetH);
  }
  
  return shades;
}
```

- Example: Yellows can shift toward green when lighter, orange when darker
- Blues can shift slightly toward cyan (lighter) or purple (darker)

**Solution (Optional):** Hue adjustment function

```javascript
function getHueAdjustment(shade, baseH) {
  // For yellows (H ≈ 60-100)
  if (baseH >= 60 && baseH <= 100) {
    if (shade <= 200) return 5;   // Shift toward green
    if (shade >= 700) return -10; // Shift toward orange
  }
  
  // For blues (H ≈ 220-260)
  if (baseH >= 220 && baseH <= 260) {
    if (shade <= 200) return 5;   // Shift toward cyan
    if (shade >= 700) return -5;  // Shift toward purple
  }
  
  return 0; // No adjustment for other hues
}
```

**Implementation:**
```javascript
const targetH = baseOklch.H + getHueAdjustment(shade, baseOklch.H);
```

**Trade-off:**
- ✅ More natural-looking scales (especially yellows/oranges)
- ❌ Adds complexity
- **Recommendation:** Ship v1 without hue shifting, add later if needed

### 3.4 Comparison: Linear Interpolation vs. Perceptual Curves

#### **Linear Interpolation (Simple but Wrong)**
```javascript
// BAD: Linear lightness distribution
const targetL = 0.1 + (shade / 1000) * 0.9; // Shade 50 → 0.145, 500 → 0.55, etc.
```

**Problem:** Perceptually uneven
- Light shades bunch together (hard to distinguish 50 from 100)
- Dark shades feel abrupt (900 vs 950 huge jump)

#### **Perceptual Curves (Better)**
```javascript
// GOOD: Exponential distribution for lighter shades
function getLightnessTarget(shade, baseL) {
  if (shade < 500) {
    // Light shades: exponential curve
    const t = shade / 500; // 0 to 1
    return 0.97 - (0.97 - baseL) * Math.pow(t, 1.5);
  } else {
    // Dark shades: power curve
    const t = (shade - 500) / 450; // 0 to 1
    return baseL * Math.pow(1 - t, 1.8);
  }
}
```

**Benefits:**
- More spacing between light shades
- Smoother transitions overall
- Matches Tailwind's actual color behavior

**Trade-off:**
- More complex than fixed targets
- **Recommendation:** Start with fixed targets (simpler), experiment with curves in v2

function renderChartsPreview(container) {
  const primary = colors[0];
  const { shades } = primary;
  
  // Bar chart data
  const data = [40, 65, 50, 80, 45];
  const maxHeight = 200;
  const barWidth = 50;
  const gap = 20;
  
  const barChartSVG = `
    <svg width="${(barWidth + gap) * data.length}" height="${maxHeight + 40}" style="display: block; margin: 0 auto;">
      ${data.map((value, i) => {
        const height = (value / 100) * maxHeight;
        const x = i * (barWidth + gap);
        const y = maxHeight - height;
        const shadeKey = [400, 500, 600, 700, 800][i];
        
        return `
          <rect x="${x}" y="${y}" width="${barWidth}" height="${height}" 
            fill="${shades[shadeKey]}" rx="4" />
          <text x="${x + barWidth / 2}" y="${maxHeight + 20}" 
            text-anchor="middle" fill="${shades[600]}" font-size="12" font-weight="700">
            ${value}%
          </text>
        `;
      }).join('')}
    </svg>
  `;
  
  // Pie chart (simple donut)
  const pieData = [30, 45, 25];
  const radius = 80;
  const centerX = 100;
  const centerY = 100;
  
  let currentAngle = 0;
  const pieSlices = pieData.map((value, i) => {
    const angle = (value / 100) * 360;
    const startAngle = currentAngle;
    const endAngle = currentAngle + angle;
    currentAngle = endAngle;
    
    const shadeKey = [500, 600, 700][i];
    
    // Arc path calculation
    const start = polarToCartesian(centerX, centerY, radius, endAngle);
    const end = polarToCartesian(centerX, centerY, radius, startAngle);
    const largeArc = angle > 180 ? 1 : 0;
    
    return `
      <path d="M ${centerX},${centerY} L ${start.x},${start.y} 
        A ${radius},${radius} 0 ${largeArc} 0 ${end.x},${end.y} Z"
        fill="${shades[shadeKey]}" />
    `;
  }).join('');
  
  const pieChartSVG = `
    <svg width="200" height="200" style="display: block; margin: 0 auto;">
      ${pieSlices}
    </svg>
  `;
  
  container.innerHTML = `
    <div class="charts-grid">
      <div class="chart-wrapper">
        <h4>Bar Chart</h4>
        ${barChartSVG}
      </div>
      <div class="chart-wrapper">
        <h4>Pie Chart</h4>
        ${pieChartSVG}
      </div>
    </div>
  `;
}

function polarToCartesian(centerX, centerY, radius, angleInDegrees) {
  const angleInRadians = (angleInDegrees - 90) * Math.PI / 180.0;
  return {
    x: centerX + (radius * Math.cos(angleInRadians)),
    y: centerY + (radius * Math.sin(angleInRadians))
  };
}

### 6.4 Figma Variables (W3C Design Tokens)

**Generate 2 files: Light mode + Dark mode (if secondary color exists)**

```javascript
function generateFigma(colors, mode = 'light') {
  const tokens = {
    "$schema": "https://design-tokens.github.io/community-group/format/",
    "$description": `Figma Variables - ${mode} mode`,
    "colors": {}
  };
  
  colors.forEach(color => {
    tokens.colors[color.name] = {};
    for (const [shade, hex] of Object.entries(color.shades)) {
      tokens.colors[color.name][shade] = {
        "$type": "color",
        "$value": hex
      };
    }
  });
  
  return JSON.stringify(tokens, null, 2);
}

function downloadFigmaFiles() {
  const primary = colors[0];
  if (primary) {
    const lightTokens = generateFigma([primary], 'light');
    downloadFile(lightTokens, 'figma-tokens-light.json');
  }
  
  const secondary = colors[1];
  if (secondary) {
    const darkTokens = generateFigma([secondary], 'dark');
    downloadFile(darkTokens, 'figma-tokens-dark.json');
  } else {
    showToast('Add a secondary color to export 2 files');
  }
}

function downloadFile(content, filename) {
  const blob = new Blob([content], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}
```

## 8. Performance Considerations

### 8.1 Real-Time Shade Calculation

**Problem:** Generating shades on every input change could lag

**Solution:** Debounce color input changes

```javascript
let colorInputTimeout;

function onColorInputChange(id, newHex) {
  clearTimeout(colorInputTimeout);
  
  colorInputTimeout = setTimeout(() => {
    updateColorBase(id, newHex);
  }, 150); // 150ms debounce
}
```

**Alternative:** Only regenerate on blur (onchange event)

```html
<input type="color" value="${color.baseHex}" 
  onchange="updateColorBase(${color.id}, this.value)">
```

**Recommendation:** Use `onchange` (simpler, no lag)

### 8.2 SVG Charts Rendering Performance

**Problem:** Complex SVG can slow down rendering

**Optimization:**
- Limit chart data points (max 10-12 bars)
- Use `transform` for positioning (faster than recalculating paths)
- Cache chart SVG if data unchanged

```javascript
let cachedChartSVG = null;

function renderChartsPreview(container) {
  if (cachedChartSVG) {
    container.innerHTML = cachedChartSVG;
    return;
  }
  
  // Generate chart SVG
  const svg = generateChartSVG();
  cachedChartSVG = svg;
  container.innerHTML = svg;
}

// Invalidate cache when colors change
function updateColorBase(id, newHex) {
  // ...
  cachedChartSVG = null; // Force re-render
  render();
}
```

### 8.3 DOM Update Strategy

**Current:** `innerHTML` bulk replacement (fast for small DOM)

**Problem:** Causes full re-render (loses input focus, scroll position)

**Solution:** Targeted updates for color input changes

```javascript
function updateShadeDisplay(colorId, shade, newHex) {
  // Find specific shade element
  const el = document.querySelector(`[data-color="${colorId}"][data-shade="${shade}"]`);
  if (el) {
    el.style.background = newHex;
    el.querySelector('.shade-hex').textContent = newHex;
  }
}
```

**Trade-off:**
- ✅ Preserves input focus
- ❌ More complex code
- **Recommendation:** Use `innerHTML` for v1 (simpler), optimize in v2 if needed

### 8.4 Memory Management

**Concern:** Large color arrays + history stack

**Mitigation:**
- Limit history to 50 states (~50KB max)
- Limit colors to 5 palettes (reasonable UX)
- Clear history on bulk clear

```javascript
function clearAll() {
  colors = [];
  history = [JSON.stringify(colors)];
  historyIndex = 0;
  render();
}
```

### 8.5 Export Generation Performance

**Problem:** Generating large export strings on every render

**Solution:** Only generate export when tab is active

```javascript
function renderExport() {
  const container = document.getElementById('exportCode');
  
  // Don't generate if export section is scrolled out of view
  const rect = container.getBoundingClientRect();
  if (rect.top > window.innerHeight + 200) {
    return; // Skip render if far off-screen
  }
  
  // Generate export based on active tab
  switch (currentExportTab) {
    case 'tailwind3': container.textContent = generateTailwind3(colors); break;
    // ...
  }
}
```

**Alternative:** Lazy generate on tab switch only

```javascript
function switchExport(btn, tab) {
  currentExportTab = tab;
  renderExport(); // Only render when tab changes
}
```

**Recommendation:** Lazy generation (avoids unnecessary work)

**Current estimate:** ~110KB uncompressed, ~28KB gzipped

**Further optimization:**

- Compress repeated CSS patterns

**Example minification:**
